\section{Motivation}
\frame{\frametitle{Overview}\tableofcontents[currentsection]}

\begin{frame}{Deep Learning requires massive datasets}
\begin{itemize}
	\item Deep Learning is today the most powerful method for image classification, segmentation and object detecction. 
   \item Deep Learning can achieve or even go beyond human performance for these tasks. 
   \item Problem: deep learning usually relies on massive annotated datasets.
   \item Example: ImageNet contains $1.4$ million annotated images \cite{ImageNet:2015}. 
\end{itemize}
%\vspace{-.5cm}
\end{frame}

\begin{frame}{Deep Learning requires massive datasets}
\begin{itemize}
\item Manual image annotation is annoying.
\item Manual image annotation is expensive.
\item Sometimes it is even impossible to get a lot of annotated data (for instance rare diseases, security relevant tasks)
\item We need to address the need for massive image data sets by:
\begin{itemize}
   \item using cheap annotations.
   \item using different (abundant) annotated data.
   \item using unlabeled data.
\end{itemize}
\end{itemize}
%\vspace{-.5cm}
\end{frame}

\section{Self-supervised Learning (SSL)}
\frame{\frametitle{Overview}\tableofcontents[currentsection]}

\begin{frame}{Self-supervised Learning (SSL): General idea}
\begin{itemize}
\item The success of Deep Learning in Computer Vision stems from the learning of good representations of the image data. 
\item Opportunity: to use unlabeled data for representation learning.
\item Problem: data without a task is not usable to train Neural Networks.
\item Idea: define a pretext task that is easily available without annotation. 
\item Self Supervision: supervised learning on labels that are directly and automatically generated from the data itself. 
\end{itemize}
\end{frame}

\begin{frame}{Early approach: training on spatial context of image patches}
\begin{figure}[htb]
   \centering
   \includegraphics[height=0.4\textheight]{../graphics/position_learning.pdf}
   \caption{Self-supervised learning using spatial context as a pretext task\cite{Doersch2015}}
\end{figure}
\begin{itemize}
   \item One patch $x_q$ is randomly sampled from the 8 positions around the central patch $x_c$.
   \item $x_c$ and $x_q$ are encoded (same network), and the encodings are fused. 
   \item The prediction task is the position of $x_q$ with respect to $x_c$.
\end{itemize}
\end{frame}

\begin{figure}[htb]

   \caption{Autoencoder: the pretext task is identical reconstruction.}
\end{figure}
\begin{itemize}
   \item Pretext task: reconstruction of the image
   \item Many variants
   \begin{itemize}
      \item denoising autoencoders: input is corrupted by noise \cite{Vincent}. 
      \item variational autoencoders: generative models based on autoencoders
   \end{itemize}
   \item Autoencoders will be treated in a later lecture. 
\end{itemize}
\end{frame}

\begin{frame}{Contrastive Learning / Siamese representation learning}
\begin{itemize}
   \item Pretext task: 
   \begin{itemize}
      \item For each image, we generate \textit{views} (i.e. transformed versions of the original image)
      \item We train a network to maximize agreement between representations of views originating from the same image. 
   \end{itemize}
   \item The representations are thus supposed to be robust with respect to the transformations. 
   \item One problem is the "collapse": a situation where all representations are 0 (trivial solution). 
   \item We will see a series of famous papers that have proposed different solutions to this problem.    
\end{itemize}
\end{frame}

\begin{frame}{A simple framework for Contrastive Learning of Visual Representations (SimCLR) \cite{Chen2020}}
\begin{figure}[htb]
   \centering
   \includegraphics[height=0.4\textheight]{../graphics/sim_clr.pdf}
   \caption{SimCLR workflow as defined in \cite{Chen2020}}
\end{figure}
\begin{itemize}
\item For each image $x$, we calculate two transformed versions $\tilde{x}_i$ and $\tilde{x}_j$ by two transformations $t$ and $t'$ drawn from a set of parametrized transformations. 
\item For $\tilde{x}_i$ and $\tilde{x}_j$, we calculate the representations $h_i$ and $h_j$, respectively. $f(\cdot)$ is the neural network we want to learn. 
\end{itemize}
\end{frame}

% SimCLR
\begin{frame}{SimCLR \cite{Chen2020}}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.4\textwidth]{../graphics/sim_clr.pdf}
   \caption{SimCLR workflow as defined in \cite{Chen2020}}
\end{figure}
\begin{itemize}
\item The representations $h_i$ and $h_j$ are mapped to a space where contrastive loss is applied, by a small neural network $g(\cdot)$. 
\item The task is thus to maximize the agreement between $z_i$ and $z_j$, i.e. we wish $z_i$ and $z_j$ to be close. 
%\item The classification task is to identify among all transformed images for each transformed image $x_i$ the transformed image $x_j$ that originates from the same image. 
\end{itemize}
\end{frame}

\begin{frame}{SimCLR \cite{Chen2020}}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.4\textwidth]{../graphics/sim_clr.pdf}
   \caption{SimCLR workflow as defined in \cite{Chen2020}}
\end{figure}
\begin{itemize}
\item We define the similarity of two representations as follows:
\begin{equation}
s(u,v) = \frac{u^Tv}{\|u\|\|v\|}
\end{equation}
\item With this, we can define the contrastive loss as follows:
 \begin{equation}
 l_{i,j} = -\log \frac{\exp \frac{s(z_i, z_j)} {\tau}} {\sum_{k=1}^{2N}\vmathbb{1}_{k\neq i}\exp \frac{s(z_i, z_k)}{\tau} } 
\end{equation}
\end{itemize}
\end{frame}

\begin{frame}{MOCO}
% \begin{figure}[htb]
%    \centering
%    \includegraphics[width=0.9\textwidth]{../graphics/.pdf}
%    \caption{Transformations used in SimCLR \cite{Chen2020}}
% \end{figure}
\begin{itemize}
   \item Momentum Contrast for Unsupervised Visual Representation Learning\cite{He2020}
   \item Contrastive Learning approach (same idea as SimCLR): 
   \begin{itemize}
      \item generate two views of an image.
      \item train a network to get encodings, where these two views are close, and those of other images are far. 
      %\item Here, we use the terms query and key images. 
   \end{itemize}
   \item Challenge: in contrastive Learning, we need large batch sizes (many negative samples, i.e. many keys): 
   \begin{itemize}
      \item Makes the problem harder, and the harder the problem the better the encodings.
      \item Negative samples can be virtually anything. So we need sample this large visual space. 
      %to model this variability by relying on many samples. 
   \end{itemize}
   \item In the following, we compare the query image representation $q$ with the key image representations $k$ (one of which comes from the same image that gave rise to $q$). 
\end{itemize}
\end{frame}

\begin{frame}{MOCO}
\begin{figure}[htb]
    \centering
    \includegraphics[width=0.8\textwidth]{../graphics/moco_settings.pdf}
    \caption{Different settings for contrastive learning \cite{He2020}}
 \end{figure}
\begin{enumerate}
   \item End-to-end: for each batch, all keys and queries are encoded on the fly, and the encoder is updated both for negative and for positive samples.
   \item Memory Bank: the representations of negative samples are not calculated online but taken from a memory bank. The encoder is updated only for the query image.  
   \item MoCo: maintaining a queue of encoded data samples, which are successively replaced by new samples, encoded with the new version of the key encoder.  
\end{enumerate}
\end{frame}

\begin{frame}{MOCO: Strategy}
\begin{itemize}
   \item The queue of keys contains (many) encoded samples. 
   \item When a mini-batch is processed: 
   \begin{itemize}
      \item two views of the each image are generated and processed (the query view and the positive key view) with the current version of the encoder. 
      \item After processing, the oldest encoded samples are removed, and the new key samples are added to the queue, serving as negatives for the next comparisons.
      \item The momentum encoder is updated with the updated query encoder:
   \begin{equation}
      \theta_k \leftarrow m\theta_k + (1-m)\theta_q
   \end{equation}
   \end{itemize}
   \item For training the contrastive task, we thus have access to a large number of negative samples, however not encoded identically.  
\end{itemize}
\end{frame}

\begin{frame}{BYOL (2020): Bootstrap Your Own Latent}
\begin{figure}[htb]
    \centering
    \includegraphics[width=0.8\textwidth]{../graphics/BYOL_workflow.pdf}
    \caption{BYOL principle \cite{Grill}}
 \end{figure}
\begin{itemize}
   \item Again, we generate two views for the same input image. 
   \item The two views are input of two different branches: the online branch and the target branch. 
   \item The design is deliberately asymmetric, between the online and the target branch. 
\end{itemize}
\end{frame}

\begin{frame}{BYOL (2020): Bootstrap Your Own Latent}
\begin{figure}[htb]
    \centering
    \includegraphics[width=0.8\textwidth]{../graphics/BYOL_workflow.pdf}
    \caption{BYOL principle \cite{Grill}}
 \end{figure}
\begin{itemize}
    \item \textbf{Online}: $y = q_\theta(g_\theta(f_\theta(v)))$ (has an additional predictor $q$).
    \item \textbf{Target}: $y' = g_{\xi}(f_{\xi}(v'))$ (no predictor). 
    \item The parameters of the target branch are not optimized, but updated by an exponentially moving average : \\
    \(\xi \leftarrow \tau \xi + (1-\tau)\theta\) .
\end{itemize}
\end{frame}

\begin{frame}{BYOL (2020): Bootstrap Your Own Latent}
\begin{itemize}
  \item \textbf{Loss}:
  \begin{equation}
   \mathcal{L}_{\theta, \xi} = \|\bar{q}_{\theta}(z_{\theta}) -\bar{z}'_{\xi}\|_2^2 = 2 - 2 \cdot \frac{\left\langle q_{\theta}(z_{\theta}),z'_{\xi} \right\rangle}{\|q_{\theta}(z_{\theta})\|_2\|z'_{\xi}\|_2} 
   \end{equation}
   \item $\bar{\cdot}$ denotes the $\ell_2$-normalized vector: $\bar{a} = a / \|a\|_2$.
   \item No explicit negatives, no contrastive denominator.
   \item The representation is trained to "chase" a slowly moving target. 
\end{itemize}
\end{frame}

\begin{frame}{BYOL: why does it not collapse?}
\begin{itemize}
\item A trivial solution exists: all images mapped to the same vector gives zero loss.
\item BYOL avoids it empirically by three ingredients:
\begin{itemize}
   \item \textbf{Stop-gradient:} the loss is only backpropagated through the online network; the target is treated as a constant.
   \item \textbf{Slowly moving target:} the target network is an EMA of the online network.
   \item \textbf{Asymmetry:} the predictor $q_\theta$ exists only on the online branch.
\end{itemize}
% \item Removing the predictor or setting $\tau = 0$ (target = online) leads to collapse \cite{Grill}.
% \item Why this works exactly is still partly an open question.
\end{itemize}
\end{frame}

\begin{figure}[htb]

    \caption{Comparison between simCLR and BYOL \cite{Grill}}
 \end{figure}
\begin{itemize}
   \item Less dependency on batch size, as no negative examples are needed. 
   \item Less dependency on the transformations: the task never becomes simple, as it can happen in simCLR
\end{itemize}
\end{frame}

\begin{frame}{DINO (2021): Self-distillation with no labels}
\begin{figure}[htb]
    \centering
    \includegraphics[width=0.4\textwidth]{../graphics/DINO_workflow.pdf}
    \caption{DINO principle \cite{Caron2021}}
 \end{figure}
 Similar architecture and same principle as BYOL\cite{Grill}: 
   \begin{itemize}
      \item Teacher-student architecture: the teacher provides the target for the student
      \item The teacher network is updated by exponentially moving average (EMA) from the student network. 
      \item No negatives (no contrastive learning).
   \end{itemize}
 \end{frame}


\begin{frame}{Principle 1/2}
% \small
% \textbf{Self-distillation without labels (DINO):}
\begin{itemize}
  \item Two networks with shared architecture: Student $g_{\theta_s}$ and Teacher $g_{\theta_t}$ (EMA of student).
  \item Like in contrastive learning, multiple views of each image are generated: 
  \begin{itemize}
   \item 2 \emph{global} views $x_1^g$, $x_2^g$ (same resolution as $x$, here $224 \times 224$).
   \item several \emph{local} views of smaller resolution (here: $96 \times 96$). 
  \end{itemize}
  \item Teacher and student generate output probability distributions over $K$ dimensions: 
  \[ P_s(x)=\mathrm{softmax}\!\left(\frac{h_{\theta_s}(f_{\theta_s}(x))}{\tau_s}\right),\quad
     P_t(x)=\mathrm{softmax}\!\left(\frac{h_{\theta_t}(f_{\theta_t}(x))}{\tau_t}\right) \]
    \item Here $f_{\theta}(\cdot)$ is the feature extractor (or backbone), and $h_{\theta}(\cdot)$ is the projection head. For downstream tasks, we only use $f_{\theta}$. The complete networks are $g = h\circ f$ for teacher and student respectively.
\end{itemize}
\end{frame}

\begin{frame}{Principle 2/2}
\begin{itemize}
  \item During training, the student aims at approaching the teacher distributions $P_t$. 
  \item Training ($V$ is the full set of views): 
  \begin{equation}
  \min_{\theta_s} \sum_{x \in \{ x_1^g, x_2^g\}} \sum_{\substack{x' \in V\\ x\neq x'}} - P_t(x)\log{P_s(x')}
  \end{equation}
  % \item The loss is the cross entropy between $P_t$ and $P_s$ : 
  % \[ \mathcal{L}=\sum_{\substack{x_1\in \text{student views}\\ x_1\in \text{teacher global views}}}
  %    \mathrm{CE}\!\left(P_t(x_2)\;\|\;P_s(x_2)\right) \]
  \item Teacher parameters are not updated by backpropagation, but via EMA: 
  \begin{equation}
   \theta _t \leftarrow \lambda \theta _t + (1-\lambda)\theta _s
  \end{equation}

\end{itemize}
\end{frame}


\begin{frame}{DINO: avoiding collapse by centering and sharpening}
\begin{itemize}
\item Two possible collapses of the teacher output $P_t$:
\begin{itemize}
   \item one dimension dominates for all images,
   \item uniform distribution for all images.
\end{itemize}
\item \textbf{Centering} (against domination): subtract a running mean $c$ from the teacher output:
\begin{align*}
P_t(x) &= \mathrm{softmax}\!\left(\frac{g_{\theta_t}(x) - c}{\tau_t}\right) \\
c &\leftarrow m\, c + (1-m)\frac{1}{B}\sum_{i=1}^B g_{\theta_t}(x_i)
\end{align*}
\item \textbf{Sharpening} (against uniformity): low teacher temperature $\tau_t < \tau_s$.
\item The two effects balance each other; together with the EMA teacher, no negatives are needed.
\item Teacher sees only the \emph{global} views, the student sees all views: ``local-to-global'' correspondence.
\end{itemize}
\end{frame}

\begin{frame}{Masked image modeling: MAE \cite{He2022}}
% \begin{figure}[htb]
%    \centering
%    \includegraphics[width=0.8\textwidth]{../graphics/MAE_workflow.pdf}
%    \caption{Masked Autoencoder \cite{He2022}}
% \end{figure}
\begin{itemize}
\item Idea from NLP (BERT): hide part of the input and predict it.
\item The image is split into patches; a large fraction ($75\%$) is \textbf{masked} at random.
\item The ViT encoder sees only the visible patches; a small decoder reconstructs the masked ones.
\item Loss on the masked patches $M$ only:
\begin{equation*}
\mathcal{L} = \frac{1}{|M|}\sum_{i \in M} \| x_i - \hat{x}_i \|_2^2
\end{equation*}
\item High masking ratio makes the task hard (no trivial interpolation) and training cheap (encoder processes $25\%$ of patches).
\item Unlike contrastive methods: no augmentations needed for the pretext task, no collapse problem.
\end{itemize}
\end{frame}

\begin{frame}{DINOv2: scaling self-supervision \cite{Oquab2024}}
\begin{itemize}
\item Combines two losses:
\begin{itemize}
   \item image-level: DINO loss on the [CLS] token,
   \item patch-level: masked patch prediction in feature space (iBOT \cite{Zhou2022}).
\end{itemize}
\item Trained on a \textbf{curated} dataset of 142M images: data quality matters as much as data quantity.
\item Large ViT (1B parameters), distilled into smaller models.
\item Frozen features (no fine-tuning) are competitive with supervised models for classification, segmentation and depth estimation.
\item DINOv3 \cite{Simeoni2025} scales this further (7B parameters) and improves dense (patch-level) features.
\end{itemize}
\end{frame}

\begin{frame}{Foundation models}
\begin{itemize}
\item \textbf{Foundation model} \cite{Bommasani2021}: a large model trained on broad data, mostly without labels, that can be adapted to many downstream tasks.
\item Adaptation with few or no labels: linear probing, fine-tuning, or zero-shot (e.g.\ with text prompts).
\item Examples in image analysis:
\begin{itemize}
   \item \textbf{General vision:} DINOv2 \cite{Oquab2024} (generic image features), CLIP \cite{Radford2021} (images and text, zero-shot classification), SAM \cite{Kirillov2023} (promptable segmentation).
   \item \textbf{Remote sensing:} SatMAE \cite{Cong2022} (MAE on multi-spectral satellite images).
   \item \textbf{Computational pathology:} UNI \cite{Chen2024}, Virchow \cite{Vorontsov2024} (DINOv2 on millions of histology tiles).
\end{itemize}
\item Common idea: pay the cost of pretraining once, then solve many tasks with few labels.
\end{itemize}
\end{frame}

\section{Weak supervision}
\frame{\frametitle{Overview}\tableofcontents[currentsection]}

\begin{frame}{What is weak supervision?}
\begin{itemize}
\item Weak supervision refers to a situation, where the ground truth data we build our model on is in some sense imperfect \cite{Zhou2018}. 
\item Three types of weak supervision:
\begin{enumerate}
\item \textbf{Incomplete supervision:} only a subset of training data are labeled (semisupervised setting).
\item \textbf{Inexact supervision:} training data are labeled but not as exactly as required by the task we would like to perform.
\item \textbf{Inaccurate supervision:} training data are labeled, but may contain mistakes. 
\end{enumerate}
\item Here, we treat the case of inexact supervision. 
\end{itemize}
\end{frame}

\begin{frame}{Multiple Instance Learning (MIL): example}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.8\textwidth]{../graphics/MIL_example_keys.pdf}
   \caption{Principle of Multiple Instance Learning. Image taken from \cite{Babenko}}
\end{figure}
\begin{itemize}
\item Key-example: we are given a number of persons each of which has several keys. You are just given the information who is able to open the door. 
\item Examples:
\begin{itemize}
   \item Video tags without frame annotation
   \item Medical imaging: image is positive in the presence of a lesion
   \item Marketing: campaign sent out to a group of people
\end{itemize}

\end{itemize}
\end{frame}

\begin{frame}{Multiple Instance Learning (MIL)}
\begin{itemize}
\item Instead of labels $y_j$ for each instance $x_j$, we only have labels for \textbf{bags} of instances:
\begin{equation*}
T = \{(B_i, Y_i)\}_{i=1}^{N}, \qquad B_i = \{x_{i1}, \ldots, x_{iK_i}\}
\end{equation*}
\item Standard MIL assumption: a bag is positive if \emph{at least one} instance is positive, negative if all instances are negative:
\begin{equation*}
Y_i = \max_k \, y_{ik}
\end{equation*}
\item The number of instances $K_i$ can vary from bag to bag.
\item The model must be \textbf{permutation invariant}: the order of instances does not matter.
\end{itemize}
\end{frame}

\begin{frame}{Aggregation functions}
Given tile embeddings $z_1, \ldots, z_K$, the slide representation $a$ can be:
\begin{itemize}
\item \textbf{Max pooling:} exactly the MIL assumption (one positive tile is enough) \cite{Campanella2019}. \\
   Works for small lesions, but only one tile receives a gradient: unstable, ignores context.
\item \textbf{Mean pooling:} $a = \frac{1}{K}\sum_k z_k$. \\
   Stable, but small relevant regions are diluted.
\item \textbf{Attention pooling:} $a = \sum_k s_k z_k$, with weights $s_k$ \emph{learned} from the tiles \cite{Ilse2018}. \\
   Compromise between max and mean: the model learns which tiles matter.
\end{itemize}
\vspace{.2cm}
All three are permutation invariant and handle a variable number of tiles.
\end{frame}

\begin{frame}{Attention-based MIL \cite{Ilse2018}}
\begin{itemize}
\item Each tile embedding $z_k$ gets a score, normalized over the bag by a softmax:
\begin{equation*}
s_k = \frac{\exp\left(w^\top \tanh(V z_k)\right)}{\sum_{j=1}^{K} \exp\left(w^\top \tanh(V z_j)\right)}
\end{equation*}
where $V$ and $w$ are learned parameters.
\item Slide representation and prediction:
\begin{equation*}
a = \sum_{k=1}^{K} s_k z_k, \qquad \hat{Y} = \sigma(c^\top a)
\end{equation*}
\item Trained end-to-end with a standard classification loss on slide labels.
\end{itemize}
\end{frame}

\item Useful for interpretation and for checking that the model looks at tissue, not artifacts.

\begin{frame}{Weak supervision in pathology: discussion}
\begin{itemize}
\item Slide-level labels come for free from clinical routine: diagnosis, molecular status, outcome.
\item This allows predicting variables that pathologists cannot see by eye (e.g.\ mutations, response to treatment).
\item Current pipelines combine the strategies of this lecture:
\begin{itemize}
   \item \textbf{self-supervision} to train the tile encoder (no labels),
   \item \textbf{weak supervision} to train the aggregation (slide labels).
\end{itemize}
\item Limitations: needs enough slides (hundreds), sensitive to confounders (center, scanner, staining): the model may learn the hospital instead of the biology $\rightarrow$ domain adaptation.
\end{itemize}
\end{frame}

\section{Domain Adaptation}
\frame{\frametitle{Overview}\tableofcontents[currentsection]}

\begin{frame}{Example: Simulation for the generation of training data}
\begin{itemize}
\item With simulation, we can generate arbitrary quantities of data with known ground truth. 
\item Simulation is regularly used for benchmarking of methods.
%\item Many different techniques: physical simulation, statistical simulation, 
\item Usually, simulation provides us with data similar to real world data, but not identical. 
\item For this reason, the use of simulated data for training neural networks is limited. 
\item Idea: can we overcome the differences between simulated and real data by explicit algorithmic strategies? 
\end{itemize}
\end{frame}

% Learning a discriminative classifier or other predictor in the presence of a shift be- tween training and test distributions is known as domain adaptation (DA). The proposed approaches build mappings between the source (training-time) and the target (test-time) domains, so that the classifier learned for the source domain can also be applied to the target domain, when composed with the learned mapping between domains.

% We consider classification tasks where X is the input space and Y = {0, 1, . . . , L−1} is the
% set of L possible labels. Moreover, we have two different distributions over X×Y , called the
% source domain DS and the target domain DT. An unsupervised domain adaptation learning
% algorithm is then provided with a labeled source sample S drawn i.i.d. from DS, and an
% unlabeled target sample T drawn i.i.d. from DX , where DX is the marginal distribution of TT
% DT over X.

\begin{frame}{Domains}
\begin{itemize}
\item Domain adaptation: learning a discriminative classifier when training data does not follow the same distribution as the test data.
\item Let $\mathcal{X}$ be an input space and $\mathcal{Y} = \{ 1, \ldots, L\}$ the set of $L$ possible labels. We call a domain the distribution over $\mathcal{X} \times \mathcal{Y}$. 
\item Here we consider two domains: the source domain $\mathcal{D}_S$ and the target domain $\mathcal{D}_T$. 
\item The $n$ source samples are then drawn from $\mathcal{D}_S$, the $N-n$ test samples from $\mathcal{D}_T$: 
\begin{eqnarray}
S &=& \{(x_i, y_i)\}_{i=1}^n \sim \mathcal{D}_S^n \nonumber \\
T &=& \{(x_i, y_i)\}_{i=n+1}^N \sim \mathcal{D}^{N-n}_T \nonumber 
\end{eqnarray}
%where $\mathcal{D}^X_T$ is the marginal distribution of $\mathcal{D}_T$. 
\end{itemize}
\end{frame}

\begin{frame}{The divergence of domains}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.7\textwidth]{../graphics/h_divergence.pdf}
   \caption{Illustration of h-divergence between domains.}
\end{figure}
\begin{itemize}
\item The divergence of domains can be quantified by trying to classify samples according to their source label. 
\item Hence, we train a classifier from a training set $T=\{(x_i, d_i)\}$: 
\begin{equation}
d_i= 
\begin{dcases}
    0,& \text{if } x_i \sim \mathcal{D}^X_S\\
    1,& \text{if } x_i \sim \mathcal{D}^X_T\\
\end{dcases}
\end{equation}
\item In this case, the h-divergence can be written as $2(1-2 \epsilon)$, where $\epsilon$ is the error of the classifier. 
\end{itemize}
\end{frame}

\begin{frame}{Domain adaptation by adversarial training}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.7\textwidth]{../graphics/domain_adaptation.pdf}
   \caption{Domain adaptation by adversarial training. Image taken from \cite{Ganin2016}}
\end{figure}
\begin{itemize}
   \item We seek a representation that provides good prediction results, but low h-divergence. 
   \item This means that this representation ($f$ in the figure) should not allow us to distinguish between source domain and target domain. 
\end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Domain adaptation by adversarial training}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.7\textwidth]{../graphics/domain_adaptation.pdf}
   \caption{Domain adaptation by adversarial training. Image taken from \cite{Ganin2016}}
\end{figure}
For this we consider the following architecture \cite{Ganin2016}: 
\begin{itemize}
   \item $G_f(;,\theta_f)$ : feature extractor
   \item $G_y(;,\theta_y)$ : label predictor 
   \item $G_d(;,\theta_d)$ : domain classifier
\end{itemize}
\end{frame}

\begin{frame}[noframenumbering]{Domain adaptation by adversarial training}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.7\textwidth]{../graphics/domain_adaptation.pdf}
   \caption{Domain adaptation by adversarial training. Image taken from \cite{Ganin2016}}
\end{figure}
We consider the following loss: 
\begin{equation}
\loss(\theta_f, \theta_y, \theta_d) = \loss_y(\theta_f, \theta_y) - \lambda \loss_d(\theta_f, \theta_d) 
\end{equation}
where $\loss_y(\theta_f, \theta_y)$ is the standard prediction loss for samples from the training set (and thus the source domain), while $\loss_d(\theta_f, \theta_d)$ is the domain loss, calculated on all samples (labeled samples from the source domain and the unlabeled samples from the target domain). 
\end{frame}

\begin{frame}{Domain adaptation by adversarial training - Optimization}
\begin{itemize}
   \item More formally, we write:
\begin{eqnarray*}
\loss_y(\theta_f, \theta_y) &=& \frac{1}{n} \sum_{i=1}^n \loss_y(G_y(G_f(x_i,\theta_f), \theta_y), y_i) \\
\loss_d(\theta_f, \theta_d) &=& \frac{1}{n} \sum_{i=1}^n \loss_d(G_d(G_f(x_i,\theta_f), \theta_d), d_i) \\
&+& \frac{1}{N-n} \sum_{i=n+1}^N\loss_d(G_d(G_f(x_i,\theta_f), \theta_d), d_i) \\
\loss(\theta_f, \theta_y, \theta_d) &=& \loss_y(\theta_f, \theta_y) - \lambda \loss_d(\theta_f, \theta_d) 
\end{eqnarray*}
\item This loss has to be minimized with respect to some parameters and maximized with respect to other parameters:
\begin{eqnarray*}
(\hat{\theta}_f, \hat{\theta}_y) &=& \argmin_{\theta_f, \theta_y} \loss(\theta_f, \theta_y, \hat{\theta}_d) \\
\hat{\theta}_d &=& \argmax_{\theta_d} \loss(\hat{\theta}_f, \hat{\theta}_y, \theta_d) \\
\end{eqnarray*}
\end{itemize}
\end{frame}

\begin{frame}{Domain adaptation by adversarial training - gradient reversal}
\begin{figure}[htb]
   \centering
   \includegraphics[width=0.7\textwidth]{../graphics/domain_adaptation.pdf}
   \caption{Domain adaptation by adversarial training. Image taken from \cite{Ganin2016}}
\end{figure}
The solution of simultaneous minimization and maximization can be elegantly solved by reversing a gradient layer during back-propagation, as indicated in the figure. 
\end{frame}

\section{Conclusion}
\frame{\frametitle{Overview}\tableofcontents[currentsection]}

\begin{frame}{Conclusion}
\begin{itemize}
\item Large annotated datasets are a bottleneck for many applications, in particular in medicine.
\item Four strategies:
\begin{enumerate}
   \item \textbf{Self-supervised learning:} representations from unlabeled data; basis of \textbf{foundation models}.
   %\item \textbf{Semi-supervised learning:} pseudo-labels and consistency on unlabeled data.
   \item \textbf{Weak supervision:} coarse labels, e.g.\ MIL for whole slide images.
   \item \textbf{Domain adaptation:} transfer from a labeled source domain to an unlabeled target domain.
\end{enumerate}
\item In practice, they are combined: e.g.\ SSL foundation model + attention MIL in pathology.
\item \textbf{Foundation models} are trained on broad unlabeled data and can be adapted to many tasks with few labels.
\end{itemize}
\end{frame}
