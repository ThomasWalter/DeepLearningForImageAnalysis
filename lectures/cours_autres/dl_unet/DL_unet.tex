\begin{frame}{Measures used in image processing}


  %::::::::::::::::::::::::::::::
  \begin{columns}
    \begin{column}{0.6\textwidth}
      Let $A$ and $B$ be two sets, not simultaneously empty ({\begin{small} Credits: public domain wikipedia.org. \end{small}})

    \end{column}
    \begin{column}{0.3\textwidth}
      \includegraphics[width=0.8\textwidth]{../images/Intersection_of_sets_A_and_B.png}
    \end{column}
  \end{columns}
  %::::::::::::::::::::::::::::::


  %::::::::::::::::::::::::::::::
  \begin{columns}
    \begin{column}{0.5\textwidth}
      \begin{block}{Jaccard index}
        \[
        J(A,B) = \frac{|A \cap B|}{|A \cup B|}
        \]
      \end{block}

    \end{column}
    \begin{column}{0.5\textwidth}
      \begin{block}{Dice coefficient}
        \[
        D(A,B) = \frac{2|A \cap B|}{|A|+|B|}
        \]
      \end{block}

    \end{column}
  \end{columns}
  %::::::::::::::::::::::::::::::

  \begin{block}{Properties}
    \begin{itemize}
    \item $\forall A, B: 0 \leq J(A, B) \leq D(A,B) \leq 1$
    \item If $A=B$, then $D(A,B) = J(A,B) = 1$
    \item If $ A \cap B = \emptyset$, then $D(A,B) = J(A,B) = 0$
    \end{itemize}
  \end{block}


\end{frame}


%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Generalization to $[0,1]$}

  \begin{block}{Jaccard index}
    Also named Intersection over Union (IoU)
        \[
        J(A,B) = \frac{|A \cap B|}{|A \cup B|}
        \]
      \end{block}

      But $\y$ and $\hat{\y}$ are in $[0,1]^n \ldots$

\pause

  %% \begin{block}{Dice similarity}
  %%   \[
  %%   D(\y, \hat{\y}) = \frac{2\sum_i y_i\hat{y}_i}{\sum_i y_i + \sum_i \hat{y}_i}
  %%   \]
  %% \end{block}

  \begin{block}{Jaccard similarity}
    \[
    J(\y, \hat{\y}) = \frac{\sum_i y_i\hat{y}_i}{\sum_i y_i + \sum_i \hat{y}_i - \sum_i y_i\hat{y}_i}
    \]
  \end{block}

($\y$ and $\hat{\y}$ are not simultaneously equal to $0$)

\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

\begin{frame}{Corresponding loss function}


  %% \begin{block}{Dice loss}
  %%   \[
  %%   d(\y, \hat{\y}) = 1-\frac{2\sum_i y_i\hat{y}_i}{\sum_i y_i + \sum_i \hat{y}_i}
  %%   \]
  %% \end{block}

  \begin{block}{Jaccard loss}
    \[
    j(\y, \hat{\y}) = \textcolor{blue}{1-}\frac{\sum_i y_i\hat{y}_i}{\sum_i y_i + \sum_i \hat{y}_i - \sum_i y_i\hat{y}_i + \epsilon}
    \]
  \end{block}

    Constant $\epsilon$, which is typically ``small'', keeps the denominator ``far enough'' from zero.


\end{frame}

\begin{frame}[fragile]{Jaccard Loss code (keras)}
\begin{verbatim}
def jaccard1_coef(y_true, y_pred, smooth=SMOOTH):
    y_true_f = K.flatten(y_true)
    y_pred_f = K.flatten(y_pred)
    intersection = K.sum(y_true_f * y_pred_f)
    union = K.sum(y_true_f) + K.sum(y_pred_f) - intersection
    return (intersection + smooth) / (union + smooth)


def jaccard1_loss(y_true, y_pred, smooth=SMOOTH):
    return 1 - jaccard1_coef(y_true, y_pred, smooth=SMOOTH)
\end{verbatim}
  \end{frame}

\begin{frame}{Cross entropy}
  Classification loss function.


  It is the sum of the true probability ($y_i$) multiplied by the log of the prediction ($\hat{y}_i$) over all classes of the distribution.

\textcolor{blue}{  \centerline{  Cross-entropy:  $CE=-\sum\limits_i y_i\log(\hat{y}_i)$}}

\begin{frame}{Softmax operator}

  \begin{block}{Definition}
    The softmax operator $\sigma: \R^d \rightarrow \R^d$ is given by:
    \[
    \forall \mathbf{x}  \in \R^d, \forall k \in \{1, \ldots, d\}:\quad \sigma(\mathbf{x})_k = \frac{e^{\mathbf{x}_k}}{\sum\limits_{i=1}^d e^{\mathbf{x}_i}}
    \]
  \end{block}

  \pause

  \begin{columns}
    \begin{column}{.3\textwidth}
      \begin{block}{Some properties}
        \begin{itemize}
        \item $0 < \sigma(\mathbf{x})_k < 1$
        \item $\sum\limits_{i=1}^d \sigma(\mathbf{x})_i = 1$
        \end{itemize}
      \end{block}

    \end{column}

    \pause

    \begin{column}{.7\textwidth}

      \begin{block}{Example}
        \[
        \mathbf{x} = \myvec{10.1\\0\\-4.3\\1.33}  \quad \sigma(\mathbf{x}) \approx \myvec{0.9998\\0.000041\\0.00000056\\0.00016}
        \]

      \end{block}



    \end{column}
  \end{columns}



\end{frame}

\begin{frame}{Cross entropy}
  \begin{block}{  Kullback - Leibler divergence}
    The dissimilarity between two probability distributions is estimated by the KL divergence.
  \end{block}
  %..........

  Consider two probability distributions $P$ and $Q$: the true probability ($P=y$) and  the predicted probability ($Q=\hat{y}_i$).

  $D_{KL}(P,Q)=\sum\limits_i P_i\log \frac{P_i}{Q_i} = \sum\limits_i P_i [\log (P_i)-\log(Q_i)]= \sum\limits_i P_i \log(P_i) - \sum\limits_i P_i \log(Q_i) $

  The first term $\sum\limits_i P_i \log(P_i)$ is the entropy of the ground truth labels, that does not depend on $\theta$. Therefore:

  $ \underset{\theta}{\mathrm{argmin}}[ D_{KL}(P,Q)] = \underset{\theta}{\mathrm{argmin}}[- \sum\limits_i P_i \log(Q_i)] \Rightarrow \textcolor{blue}{\text{Cross Entropy formula}}$
%https://www.youtube.com/watch?v=SxGYPqCgJWM
\end{frame}

\begin{frame}{Focal loss}
  %..........
  \begin{block}{Goal}
Designed to fight against class imbalance, focuses training on a
sparse set of "hard" examples and prevents the vast number of "easy"
samples from overwhelming the detector.
    \end{block}
      $$
      CE(p,y)=
    \left\lbrace
      \begin{array}{cc}
        -log(p)  & \text{if } y = 1\\
        -log(1-p) & otherwise
      \end{array}\right.
      $$

      y: ground truth class

      $p \in [0,1]$: estimated probability for the class y = 1

            $$
      p_t=
    \left\lbrace
      \begin{array}{cc}
        p  & \text{if } y = 1\\
        (1-p) & otherwise
      \end{array}\right.
      $$

      $$ CE(p,y)=CE(p_t)=-log(p_t)$$

      {\scriptsize   Focal Loss for Dense Object Detection
  https://arxiv.org/pdf/1708.02002v2.pdf
}
\end{frame}

\begin{frame}{Optimizers}

  The problem is strongly non-convex. We do not have convergence guarantees. Only a few guidelines.

  \item Stochastic Gradient Descent (SGD)

\begin{frame}[fragile]{SGD + momentum}

  %..........
  \begin{block}{Momentum}
     Accelerates optimization, integrating gradient with an exponentially decaying moving average


    \end{block}

  \begin{alltt}
    v = 0
    for i in nb_steps:
        dw = compute_gradient(lossF,data,w)
        \textcolor{blue}{v = rho * v + dw}
        w = w - lr * \textcolor{blue}{v}
  \end{alltt}
  %..............................
  \begin{itemize}{}
  \item Intuition: a ball going down a hill. The ball does not follow the gradient direction because of its velocity.
  \item $\rho$ typically 0.9 or 0.99
  \end{itemize}

\begin{frame}[fragile]{SGD + momentum}

  $$ v_i = \rho v_{i-1} + dw_i$$
    $ v_0 = dw_0$

  $v_1 = \rho * dw_0 + dw_1$

  $v_2 = \rho * v_1 + dw_2 = \rho (\rho * v_0 + dw_1)+ dw_2 = \rho^2 dw_0 + \rho dw_1 + dw_2$

  $...$

  $ v_i =\rho^i dw_0 + \rho^{i-1} dw_1 + ....+  dw_i $

\textcolor{blue}{    $$ v_i =\rho^i dw_0 + \rho^{i-1} dw_1 + ....+  dw_i $$}

%..............................
\begin{itemize}{}

    \item If gradient is in the same direction it is amplified

    \item Abrupt changes are smoothed


\end{itemize}
%..............................

{\scriptsize Sutskever et al."On the importance of initialization and momentun in DL" ICML 2013.}
\end{frame}

  \item  If the gradient is high in one dimension and low in another one:

    \item The optimization process may be slow, if the learning rate is small

      In both cases, too many iterarations are required for convergence, if it converges!

\begin{frame}[fragile]{AdaGrad}
\textcolor{blue}{  Adaptive} learning rates (per parameter).
  \begin{alltt}
    grad_sq = 0
    for i in nb_steps:
        dw = compute_gradient(lossF,data,w)
        \textcolor{blue}{grad_sq +=  dw * dw}
        w = w - lr*dw/(grad_sq+epsilon)
    \end{alltt}

  The squared gradient is always positive. Its sum is increasing over time. Then, AdaGrad decreases the learning rate over time.

  The square gradient may become too big. The optimization progress may stop.

  Duchi et al. "Adaptive subgradient methods for online learning and stochastic optimization" JMLR 2011.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{RMSProp: "Leak Adagrad"}
  \begin{alltt}
    grad_sq = 0
    for i in nb_steps:
        dw = compute_gradient(lossF,data,w)
        \textcolor{blue}{grad_sq = decay_rate * grad_sq + (1-decay_rate) * dw * dw}
        w = w - lr*dw/(grad_sq+epsilon)
    \end{alltt}

  Intuition: do not slow down over the optimization process.

  Momentum based optimization can overshoot and then come back.

  AdaGrad and RMSProp arrest progress along the fast moving direction while simultaneously accelerating progress along slow moving direction. $\Rightarrow$ Bends directly to the minimum.
\end{frame}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{Adam: RMSProp + momentum}
  Bias at the beginning of the optimization process, gradient is close to 0.


  \begin{alltt}
    grad_sq = 0
    for i in nb_steps:
        dw = compute_gradient(lossF,data,w)
        moment1 = beta1 * moment1 + (1-beta1) * dw
        moment2 = beta2 * moment2 + (1-beta2) * dw * dw

        momment1_unbias = moment1/(1-beta1**i)
        momment2_unbias = moment2/(1-beta2**i)
        w = w - lr*moment1_unbias/(moment2_unbias+epsilon)

    \end{alltt}


default values: $ \beta_1=0.9; \beta_2 = 0.999$

  Kingma and Ba, "Adam: A method for stochastic optimization", ICLR 2015.
\end{frame}

\begin{frame}{Weigth Decay}

{\bf  Idea}: keep weights small unless absolutely necessary. A regularization technique that prevents overfitting.

  %::::::::::::::::::::::::::::::

    \begin{column}{0.35\textwidth}
        $$ \mathcal{L}_{total} = \mathcal{L}_{original}+ \lambda \sum w_i^2$$

    \end{column}
    \begin{column}{0.65\textwidth}
        \begin{itemize}{}
        \item

          $ \mathcal{L}_{original}$: Original loss, i.e. cross entropy, dice...
        \item $ \sum w_i^2$ : $L_2$ norm of all weights
        \item $\lambda$: weight decay coefficient
        \end{itemize}
    \end{column}


  \end{columns}

\subsection{Receptive field}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\frame{
  \frametitle{Receptive field}

  \begin{block}{Definition: links between neurons}
    In a NN, we say that neuron $a$ is linked to neuron $b$ if there is an oriented path in the corresponding graph going from $a$ to $b$.
  \end{block}


  \begin{block}{Definition}
    The \alert{receptive field} of a neuron in a NN is the set of \emph{input neurons} that are linked to that neuron.

    The size of the receptive field is an essential property when designing a fully-convolutional NN architecture.
  \end{block}

Note that in most cases, receptive fields are square shaped (border effects ignored).

}

\begin{frame}{U-net architecture}

  %..........
  \begin{block}{Architecture}
  \begin{itemize}{}
  \item Fully convolutional network, with an encoder and a decoder.

  \item The \textbf{encoder} (contracting path) is inspired by typical classification networks:
    %..............................
    \begin{itemize}{}
    \item    a series basic blocks: Convolution (3x3), Activation (RELU) and max-pooling 2x2 for downsampling.
  \item At each downsampling step, the number of channels (features) is multiplied by 2.
    \end{itemize}
    %..............................
  \item The \textbf{decoder} (expanding path)
    %..............................
    \begin{itemize}{}
    \item upsampling the feature map
    \item concatenate corresponding features from contracting path $\Rightarrow$   \textcolor{blue}{\underline{\textbf{ Skip connection}}}
    \item Basic block: Convolution (3x3), Activation (RELU)
    \end{itemize}
    %..............................
  \item A final Convolution 1x1 with the output number of classes.
  \end{itemize}
  %..............................

    \end{block}
  %..............................

{\small    Ronneberger, O., Fischer, P., and Brox, T. (\textbf{2015}). U-net: Convolutional networks for biomedical image segmentation. MICCAI 2015.} University of Freiburg, Germany.

  \end{frame}

\begin{frame}{Skip connection}
  %..........
  \begin{block}{The key idea}

  Skip connection combine:
  %..............................
  \begin{itemize}{}
  \item  spatial information from the encoding part
  \item  with up-sampled deeper layers with more relevant feature representations.
  \end{itemize}
  %..............................

    \end{block}

\end{frame}

    \begin{alltt}
  def \textcolor{blue}{modular_u_net}(shape, nb_levels = 4, nb_filters=32,output_channels=1):
    x = Input(shape); first_layer=[]; first_layer.append(x); skip_layers=[]
    \textcolor{blue}{# ENCODER}
    for lev in range(nb_levels):
        x = unet_block(nb_filters, res, "conv"+str(lev+1), batch_norm)(x)
        if(lev < nb_levels -1):#not the last one
            skip_layers.append(x)
            x = unet_down(conv_sampling, nb_filters)(x)
            nb_filters = nb_filters * 2
    \textcolor{blue}{# DECODER}
    for lev in range(nb_levels-1):
        x = unet_up(conv_sampling, nb_filters)(x)
        \textcolor{blue}{\textbf{x = concatenate([x, skip_layers.pop()], axis=3)}}
        nb_filters= int(nb_filters/2)
        x = unet_block(nb_filters, res, "conv"+str(nb_levels+lev+1), batch_norm)(x)
    xFinal = Conv2D(output_channels, 1, activation="sigmoid", name="out")(x)
    return Model(first_layer.pop(), xFinal)
\end{alltt}
  \end{small}
  \end{frame}

\begin{frame}[fragile]{U-net block}

  %..........
  \begin{alltt}
    def \textcolor{blue}{unet_block}(nb_fil, res, name, batch_norm):
    def fn(tensor):
        x = Conv2D(nb_fil, 3, padding="same", name="\%s_1" \% name)(tensor)
        if batch_norm:
            x = BatchNormalization()(x)
        x = Activation("relu")(x)
        x = Conv2D(nb_fil, 3, padding="same", name="\%s_2" \% name)(x)
        if batch_norm:
            x = BatchNormalization()(x)
        if res:
            skip=Conv2D(nb_fil,3,padding="same",name="\%s_res"\% name)(tensor)
            x = add([x, skip])
        x = Activation("relu")(x)
        return x
    return fn
    \end{alltt}
  \end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}[fragile]{U-net down block}
  %..........
  \begin{alltt}
def \textcolor{blue}{unet_down}(conv_sampling, nb_filters=None):
    if conv_sampling:
        if nb_filters is None:
            raise ValueError("When conv_sampling is True, nb_filters should be given")
        def fn(tensor):
            x = Conv2D(nb_filters, 3, padding="same", strides=(2, 2))(tensor)
            return x
    else:
        def fn(tensor):
            x = MaxPooling2D(pool_size=(2, 2))(tensor)
            return x
    return fn

    \end{alltt}
  \end{frame}

\begin{frame}[fragile]{U-net up block}
  %..........
  \begin{alltt}
def \textcolor{blue}{unet_up}(conv_sampling, nb_filters):
    if conv_sampling:
        if nb_filters is None:
            raise ValueError("When conv_sampling is True, nb_filters should be given")
        def fn(tensor):
            x = Conv2DTranspose(nb_filters, 3, padding="same", strides=(2, 2))(tensor)
            return x
    else:
        def fn(tensor):
            x = UpSampling2D(size=(2, 2))(tensor)
            return x
    return fn

    \end{alltt}
  \end{frame}

\begin{frame}[fragile]{Attention gate code}
%  \centerline{  \includegraphics[width=0.5\textwidth]{../images/unet_attention_gate.png}}
  \begin{small}
%https://www.youtube.com/watch?v=KOF38xAvo8I
\begin{verbatim}
def attention_block_2d(x, g, inter_channel):
    # x is the skip connexion
    x2 = Conv2D(inter_channel, [1, 1], strides=[1, 1])(x)
    # g is the up-sampled deeper level feature (stride is then 1)
    g2 = Conv2D(inter_channel, [1, 1], strides=[1, 1])(g)

    # add both branches and then activation
    f = Activation('relu')(add([x2, g2]))

    # get the scalar value, then apply sigmoid
    psi_f = Conv2D(1, [1, 1], strides=[1, 1])(f)
    rate = Activation('sigmoid')(psi_f)

    # multiply
    att_x = multiply([x, rate])
    return att_x


\end{verbatim}
\end{small}
